52# Copies one of the caller's gallery items into a diagram's canvas, the server-side twin of the 

53# editor's Gallery panel: the item keeps its layout, gets new ids and lands where asked without 

54# covering anything. The change is saved like any other edit: recorded in the history under its 

55# author and pushed to the editors that have the diagram open. 

56class InsertGalleryItem(Usecase[InsertGalleryItemParams, GalleryInsertion]): 

57 def __init__( 

58 self, 

59 items: GalleryItemRepository, 

60 diagrams: DiagramRepository, 

61 notifier: DiagramUpdateNotifier, 

62 recorder: RevisionRecorder, 

63 limits: GalleryLimits, 

64 new_id: Callable[[], str] = _random_id, 

65 ) -> None: 

66 self._items = items 

67 self._diagrams = diagrams 

68 self._notifier = notifier 

69 self._recorder = recorder 

70 self._limits = limits 

71 self._new_id = new_id

73 async def execute(self, params: InsertGalleryItemParams) -> GalleryInsertion: 

74 if not CANVAS_MIN_SCALE <= params.scale <= CANVAS_MAX_SCALE: 

75 raise InvalidInputError( 

76 f"The scale must be between {CANVAS_MIN_SCALE} and {CANVAS_MAX_SCALE}" 

77 ) 

78 item = await get_owned_gallery_item(self._items, params.item_id, params.owner_id) 

79 diagram = await self._diagrams.get_by_id(params.diagram_id) 

80 if diagram is None: 

81 raise NotFoundError(f"Diagram {params.diagram_id} not found") 

82 canvas = diagram.canvas_state 

83 if not canvas or "schema" not in canvas: 

84 raise InvalidInputError( 

85 "The diagram has no canvas yet: open it once in the editor, then insert again" 

86 ) 

87 insertion = CanvasInsertion(canvas, self._new_id) 

88 plan = _plan(insertion, item, params) 

89 merged = self._merged_canvas(canvas, insertion, plan) 

90 updated = await self._save(diagram, merged, params, item) 

91 return GalleryInsertion( 

92 diagram=updated, 

93 item_id=item.id, 

94 created_ids=plan.created_ids, 

95 root_shape_ids=plan.root_shape_ids, 

96 x=round(plan.bounds.min_x, 2), 

97 y=round(plan.bounds.min_y, 2), 

98 width=round(plan.bounds.width, 2), 

99 height=round(plan.bounds.height, 2), 

100 )

102 def _merged_canvas( 

103 self, canvas: dict[str, Any], insertion: CanvasInsertion, plan: InsertionPlan 

104 ) -> dict[str, Any]: 

105 _ensure_room_for(insertion, plan) 

106 store = dict(canvas.get("store") or {}) 

107 store.update({str(record["id"]): record for record in plan.records}) 

108 merged = canvas | {"store": store} 

109 self._ensure_fits(merged) 

110 return merged

112 def _ensure_fits(self, canvas: dict[str, Any]) -> None: 

113 size = len(json.dumps(canvas).encode()) 

114 if size > self._limits.max_canvas_bytes: 

115 raise PayloadTooLargeError( 

116 f"The diagram would grow to {size} bytes (limit {self._limits.max_canvas_bytes})" 

117 )

119 async def _save( 

120 self, 

121 diagram: Diagram, 

122 canvas: dict[str, Any], 

123 params: InsertGalleryItemParams, 

124 item: GalleryItem, 

125 ) -> Diagram: 

126 before = DiagramSnapshot.of(diagram) 

127 diagram.canvas_state = canvas 

128 updated = await self._diagrams.update(diagram) 

129 summary = params.revision_summary or f"Inserted “{item.name}” from the gallery" 

130 await self._recorder.record( 

131 updated.id, 

132 before, 

133 DiagramSnapshot.of(updated), 

134 params.author, 

135 summary=summary[:REVISION_SUMMARY_MAX_LENGTH], 

136 ) 

137 # No origin tab: every open editor, the requester's included, must load the new shapes. 

138 await self._notifier.notify_updated(updated, None) 

139 return updated

143 insertion: CanvasInsertion, item: GalleryItem, params: InsertGalleryItemParams

152 if item.image_data is None or item.image_mime_type is None:

162def _ensure_room_for(insertion: CanvasInsertion, plan: InsertionPlan) -> None: 

163 new_shapes = sum(1 for record in plan.records if record["typeName"] == "shape") 

164 if insertion.shape_count() + new_shapes > CANVAS_MAX_SHAPES_PER_PAGE:
